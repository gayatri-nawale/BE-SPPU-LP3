import numpy as np
import matplotlib.pyplot as plt

def f(x):
	return (x+3)**2

def gradient(x):
	return 2*(x+3)

x=2
iterations=100
tolerance=0.000001
learningrate=0.1

path=[x]

for i in range(iterations):
	new_x=x-learningrate * gradient(x)
	path.append(new_x)

	if(abs(new_x-x)<tolerance):
		x=new_x
		break
	x=new_x

print("Minimum x : ",round(x,3))
print("Minimum y: ",round(f(x),3))
print("Iterations : ",(i+1))


# plot function
X=np.linspace(-8,4,100)

plt.plot(X,f(X),label="y=(x+3)^2")

plt.scatter(2,f(2),s=100,color='red',label='Starting point')
plt.scatter(x,f(x),s=100,color='blue',label='Minimum point')

plt.plot(path,[f(v) for v in path],'o--',label='Gradient Descent Graph')

plt.xlabel('x axis')
plt.ylabel('y axis')
plt.grid()
plt.legend()
plt.show()

import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.neighbors import KNeighborsClassifier
from sklearn.metrics import accuracy_score, recall_score, confusion_matrix, precision_score

df=pd.read_csv('diabetes.csv')
df.head()


df.isnull().sum()

df.duplicated().sum()

df.shape


print((df[['Glucose','BloodPressure','SkinThickness','Insulin','BMI']]==0).sum())

impcol=['Glucose','BloodPressure','SkinThickness','Insulin','BMI']
for c in impcol:
    df[c]=df[c].replace(0,np.nan)
    df[c]=df[c].fillna(df[c].median())

x=df.drop(columns=['Outcome'])
y=df['Outcome']

x_train,x_test,y_train,y_test=train_test_split(x,y,test_size=0.2,random_state=42, stratify=y)

# feature scaling
scaler=StandardScaler()
x_train=scaler.fit_transform(x_train)
x_test=scaler.transform(x_test)

# knn
knn=KNeighborsClassifier(n_neighbors=5)
knn.fit(x_train,y_train)

y_pred_knn=knn.predict(x_test)

# evalution
cm=confusion_matrix(y_test,y_pred_knn)
tn,fp,fn,tp=cm.ravel()

print("Accuracy: ",accuracy_score(y_test,y_pred_knn))
print("Recall: ",recall_score(y_test,y_pred_knn))
print("Error Rate : ", 1 - accuracy_score(y_test,y_pred_knn))
print("Precision: ",precision_score(y_test,y_pred_knn))

print("True Positive: ",tp)
print("True Negative: ",tn)
print("False Positive: ",fp)
print("False Negative: ",fn)
